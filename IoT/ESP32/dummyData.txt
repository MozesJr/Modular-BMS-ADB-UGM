/*
  ============================================================================
  BMS Dummy Display — ESP32 + ILI9341 (320x240) + 6 tombol
  Untuk uji vibrasi enclosure Modular Universal BMS.
  ============================================================================
  Data dummy = snapshot JK-BMS asli via UART (04/11/2026 15:03):
    24S LiFePO4, 79.02 V, 0.0 A, SOC 69 %, kapasitas 30 Ah, cycles 0,
    MOS 24 C, T1/T2 tidak terbaca (probe NTC belum terpasang),
    CHG ON, DSG ON, balancing OFF, delta cell 8 mV.

  Library (Arduino Library Manager):
    - Adafruit ILI9341
    - Adafruit GFX Library
  Board: "ESP32 Dev Module" (esp32 core 2.x / 3.x)

  Mapping tombol (sesuai bentuk lubang di enclosure):
    ▲  UP      : halaman sebelumnya / cell sebelumnya (tahan = auto-repeat)
    ▼  DOWN    : halaman berikutnya / cell berikutnya (tahan = auto-repeat)
    ▬  ENTER   : masuk mode pilih cell -> detail cell
    ✕  BACK    : kembali satu level
    □  RESET   : tekan = reset statistik min/max + ke Summary
                 tahan 3 dtk = reset juga boot counter (NVS)
    ○  RESTART : tahan 2 dtk = ESP.restart()  (tekan singkat tidak ngapa-ngapain,
                 supaya getaran tidak memicu restart)

  Halaman: Summary -> Cells (grid 24) -> Chart -> Proteksi -> Sistem
  ============================================================================
*/

const int CELL_COUNT = 24;
const uint16_t SNAP_CELL_MV[CELL_COUNT] = {
  3294, 3294, 3294, 3294, 3294, 3294, 3294, 3294, 3294, 3294, 3294, 3294,

struct BmsSettings {
  float cellOvp = 3.650, cellOvpRec = 3.450, cellUvp = 2.500, cellUvpRec = 2.900;
  float packOvp = 87.60, packUvp = 60.00, balStart = 3.000;
  uint16_t diffProtMv = 300, balTrigMv = 10;
  float ocpDsg = 80, ocpChg = 80, capacityAh = 30;
} SET;

struct Bms {
  uint16_t cellMv[CELL_COUNT];
  float current = 0.0;
  uint8_t soc = 69;
  uint16_t cycles = 0;
  float mosTemp = 24;
  float t1 = NAN, t2 = NAN;
  bool chg = true, dsg = true, bal = false;
  const char* fw = "S23.06";
  // turunan
  float packV = 0, avgMv = 0;
  uint16_t maxMv = 0, minMv = 0;
  uint8_t maxIdx = 0, minIdx = 0;
} bms;

uint16_t seenMin[CELL_COUNT], seenMax[CELL_COUNT];

void computeDerived() {
  uint32_t sum = 0;
  bms.maxMv = 0; bms.minMv = 65535;
  for (int i = 0; i < CELL_COUNT; i++) {
    uint16_t v = bms.cellMv[i];
    sum += v;
    if (v > bms.maxMv) { bms.maxMv = v; bms.maxIdx = i; }
    if (v < bms.minMv) { bms.minMv = v; bms.minIdx = i; }
    if (v < seenMin[i]) seenMin[i] = v;
    if (v > seenMax[i]) seenMax[i] = v;
  }
  bms.packV = sum / 1000.0f;
  bms.avgMv = (float)sum / CELL_COUNT;
}

void resetStats() {
  for (int i = 0; i < CELL_COUNT; i++) { seenMin[i] = 65535; seenMax[i] = 0; }
  computeDerived();
}

void tickData() {
  for (int i = 0; i < CELL_COUNT; i++) {
    int j = DUMMY_JITTER_MV > 0 ? (int)random(-DUMMY_JITTER_MV, DUMMY_JITTER_MV + 1) : 0;
    bms.cellMv[i] = SNAP_CELL_MV[i] + j;
  }
  computeDerived();
}

// =============================== TOMBOL =====================================
enum BtnId { B_UP, B_DOWN, B_ENTER, B_BACK, B_RESET, B_RESTART, B_COUNT };
const char* BTN_NAME[B_COUNT]  = {"UP", "DOWN", "ENTER", "BACK", "RESET", "RESTART"};
const char* BTN_SHORT[B_COUNT] = {"\x1E", "\x1F", "ENT", "BCK", "RST", "RBT"};
const uint8_t BTN_PIN[B_COUNT] = {PIN_BTN_UP, PIN_BTN_DOWN, PIN_BTN_ENTER,
                                  PIN_BTN_BACK, PIN_BTN_RESET, PIN_BTN_RESTART};

enum BtnEvt { EV_DOWN, EV_REPEAT, EV_LONG, EV_UP };

struct Button {
  bool stable = false, raw = false, longDone = false, swallow = false;
  uint32_t lastEdge = 0, downAt = 0, nextRepeat = 0, lastPressAt = 0;
  uint32_t count = 0;
} btn[B_COUNT];

void handleButton(BtnId id, BtnEvt ev, uint32_t heldMs);

void pollButtons(uint32_t now) {
  for (int i = 0; i < B_COUNT; i++) {
    Button& b = btn[i];
    bool r = digitalRead(BTN_PIN[i]) == LOW;
    if (r != b.raw) { b.raw = r; b.lastEdge = now; }
    if (r != b.stable && now - b.lastEdge >= DEBOUNCE_MS) {
      b.stable = r;
      if (r) {
        b.downAt = now; b.longDone = false; b.swallow = false;
        b.nextRepeat = now + REPEAT_DELAY_MS;
        b.count++; b.lastPressAt = now;
        handleButton((BtnId)i, EV_DOWN, 0);
      } else {
        handleButton((BtnId)i, EV_UP, now - b.downAt);
      }
    }
    if (b.stable) {
      uint32_t longMs = (i == B_RESTART) ? LONG_RESTART_MS : LONG_RESET_MS;
      if (!b.longDone && now - b.downAt >= longMs) {
        b.longDone = true;
        handleButton((BtnId)i, EV_LONG, now - b.downAt);
      }
      if (now >= b.nextRepeat) {
        b.nextRepeat = now + REPEAT_RATE_MS;
        handleButton((BtnId)i, EV_REPEAT, now - b.downAt);
      }
    }
  }
}

void txt(int x, int y, uint8_t sz, uint16_t fg, uint16_t bg, const char* s) {
  tft.setTextSize(sz);

// Teks rata kiri, di-pad spasi sampai `pad` karakter (menimpa teks lama tanpa flicker)
void txtf(int x, int y, uint8_t sz, uint16_t fg, uint16_t bg, int pad, const char* fmt, ...) {
  char buf[64];
  va_list ap; va_start(ap, fmt); vsnprintf(buf, sizeof(buf), fmt, ap); va_end(ap);
  int n = strlen(buf);
  while (n < pad && n < (int)sizeof(buf) - 1) buf[n++] = ' ';
  buf[n] = 0;
  txt(x, y, sz, fg, bg, buf);
}

// Teks rata kanan dengan lebar tetap `pad` karakter, berakhir di xr
void txtR(int xr, int y, uint8_t sz, uint16_t fg, uint16_t bg, int pad, const char* fmt, ...) {
  char raw[48], buf[64];
  va_list ap; va_start(ap, fmt); vsnprintf(raw, sizeof(raw), fmt, ap); va_end(ap);
  int n = strlen(raw);
  int sp = pad > n ? pad - n : 0;
  memset(buf, ' ', sp);
  memcpy(buf + sp, raw, n + 1);
  txt(xr - (sp + n) * 6 * sz, y, sz, fg, bg, buf);
}

void fmtUptime(char* out, size_t len, uint32_t ms) {
  uint32_t s = ms / 1000;
  snprintf(out, len, "%02lu:%02lu:%02lu", (unsigned long)(s / 3600),
           (unsigned long)((s / 60) % 60), (unsigned long)(s % 60));
}

void fmtTemp(char* out, size_t len, float t) {
  if (isnan(t)) snprintf(out, len, "--");
  else snprintf(out, len, "%.0f\xF8" "C", t);
}

  strncpy(toastMsg, msg, sizeof(toastMsg) - 1);

// ---- Cells grid ----
const int GRID_COLS = 4;
int gridRows() { return (CELL_COUNT + GRID_COLS - 1) / GRID_COLS; }

void cellBoxRect(int i, int& x, int& y, int& w, int& h) {
  int top = 26, bottom = H - 18;
  w = W / GRID_COLS;
  h = (bottom - top) / gridRows();
  x = (i % GRID_COLS) * w;
  y = top + (i / GRID_COLS) * h;
}

void drawCellBox(int i, bool full) {
  int x, y, w, h; cellBoxRect(i, x, y, w, h);
  bool sel = (mode == M_SELECT && i == cursor);
  if (full) {
    tft.fillRect(x + 1, y + 1, w - 2, h - 2, C_BG);
    tft.drawRect(x + 1, y + 1, w - 2, h - 2, sel ? C_SEL : C_PANEL);
    if (sel) tft.drawRect(x + 2, y + 2, w - 4, h - 4, C_SEL);
    txtf(x + 5, y + 4, 1, sel ? C_SEL : C_DIM, C_BG, 3, "C%02d", i + 1);
  }
  const char* tag = "   ";
  uint16_t col = C_TXT;
  if (i == bms.maxIdx) { tag = "MAX"; col = C_MAXC; }
  if (i == bms.minIdx) { tag = "MIN"; col = C_MINC; }
  txt(x + w - 24, y + 4, 1, col, C_BG, tag);
  txtf(x + 8, y + 14, 2, col, C_BG, 5, "%.3f", bms.cellMv[i] / 1000.0);
}

void drawCellsStatic() {
  for (int i = 0; i < CELL_COUNT; i++) drawCellBox(i, true);
}
void drawCellsDynamic() {
  for (int i = 0; i < CELL_COUNT; i++) drawCellBox(i, false);
}

  float lo = SET.cellUvp, hi = SET.cellOvp;
  auto px = [&](float v) { return bx + (int)((v - lo) / (hi - lo) * bw); };
  tft.fillRect(bx, by, bw, bh, C_PANEL);
  int p = constrain(px(mv / 1000.0), bx, bx + bw - 3);
  tft.fillRect(bx, by + 4, p - bx, bh - 8, C_ACC);
  tft.fillRect(p, by - 3, 3, bh + 6, col);

  tft.drawFastVLine(pb, by, bh, C_WARN);
  tft.drawFastVLine(pr, by, bh, C_BAD);

  txtf(bx, by + bh + 4, 1, C_DIM, C_BG, 0, "%.2f", lo);

void drawChartDynamic() {
  // skala di-zoom supaya selisih beberapa mV kelihatan
  int lo = bms.minMv - 5, hi = bms.maxMv + 5;
  if (hi - lo < 20) { int m = (hi + lo) / 2; lo = m - 10; hi = m + 10; }
  txtR(CH_X - 4, CH_Y, 1, C_DIM, C_BG, 5, "%.3f", hi / 1000.0);
  txtR(CH_X - 4, CH_Y + chH() - 8, 1, C_DIM, C_BG, 5, "%.3f", lo / 1000.0);
  int bw = chW() / CELL_COUNT, ch = chH();
  for (int i = 0; i < CELL_COUNT; i++) {
    int x = CH_X + i * bw + 1;
    int bh = (int)((long)(bms.cellMv[i] - lo) * ch / (hi - lo));
    bh = constrain(bh, 1, ch);
    uint16_t c = (i == bms.maxIdx) ? C_MAXC : (i == bms.minIdx) ? C_MINC : C_ACC;
    tft.fillRect(x, CH_Y, bw - 2, ch - bh, C_BG);
    tft.fillRect(x, CH_Y + ch - bh, bw - 2, bh, c);
  }
  int ya = CH_Y + ch - (int)((bms.avgMv - lo) * ch / (hi - lo));
  for (int x = CH_X; x < CH_X + chW(); x += 6) tft.drawFastHLine(x, ya, 3, C_TXT);

// ============================== SCREENSAVER =================================
// Mata robot + orang joget, sama seperti versi Raspi.
enum SaverKind { SV_EYES, SV_DANCE };
SaverKind saverKind = SV_EYES;
uint32_t saverStart = 0, saverFrameAt = 0, saverSwitchAt = 0;

// --- mata ---
struct EyeState { int lookX = 0, lookY = 0, open = 100; } eye, eyeDrawn;
uint32_t nextSaccade = 0, blinkAt = 0;
int blinkPhase = -1;

void drawEyes(const EyeState& e, uint16_t color) {
  const int ew = 70, eh = 84, gap = 40;
  int h = max(6, eh * e.open / 100);
  for (int s = -1; s <= 1; s += 2) {
    int cx = W / 2 + s * (ew / 2 + gap / 2) + e.lookX;
    int cy = H / 2 + e.lookY;
    tft.fillRoundRect(cx - ew / 2, cy - h / 2, ew, h, min(16, h / 2), color);
  }
}

void saverEyesFrame(uint32_t now) {
  if (now >= nextSaccade) {
    static const int pos[][2] = {{0,0},{-28,0},{28,0},{0,-14},{-20,10},{20,10},{0,0}};
    int k = random(0, 7);
    eye.lookX = pos[k][0]; eye.lookY = pos[k][1];
    nextSaccade = now + random(900, 2600);
  }
  if (blinkPhase < 0 && now >= blinkAt) blinkPhase = 0;
  if (blinkPhase >= 0) {
    static const int seq[] = {60, 20, 5, 20, 60, 100};
    eye.open = seq[blinkPhase++];
    if (blinkPhase >= 6) { blinkPhase = -1; blinkAt = now + random(2500, 5500); }
  }
  if (memcmp(&eye, &eyeDrawn, sizeof(eye)) != 0) {
    drawEyes(eyeDrawn, C_BG);
    drawEyes(eye, C_ACC);
    eyeDrawn = eye;
  }
}

// --- joget (stick figure) ---
struct Pose { int16_t la, ra, ll, rl, lean, bob; };  // sudut derajat (0 = ke bawah)
const Pose POSES[] = {
  { 150,  -150,  20, -20,  0,  0},
  { 100,   -30,  35, -10,  6, -4},
  {  30,  -100,  10, -35, -6, -4},
  { 170,  -170,  25, -25,  0, -8},
  {  60,   -60,  40,   0,  4,  0},
  {  90,   -90,   0, -40, -4,  0},
};
const int POSE_N = sizeof(POSES) / sizeof(POSES[0]);
int poseIdx = 0, poseDrawn = -1;

void thickLine(int x0, int y0, int x1, int y1, uint16_t c) {
  tft.drawLine(x0, y0, x1, y1, c);
  tft.drawLine(x0 + 1, y0, x1 + 1, y1, c);
  tft.drawLine(x0, y0 + 1, x1, y1 + 1, c);
}

void drawStick(int idx, uint16_t c) {
  const Pose& p = POSES[idx];
  int cx = W / 2 + p.lean, top = 60 + p.bob;
  int neckY = top + 30, hipY = top + 90;
  tft.drawCircle(cx, top + 14, 14, c);
  tft.drawCircle(cx, top + 14, 13, c);
  thickLine(cx, neckY, cx - p.lean / 2, hipY, c);
  int sx = cx - p.lean / 4, sy = neckY + 12;
  float la = radians(p.la), ra = radians(p.ra);
  thickLine(sx, sy, sx + (int)(sin(la) * -40), sy + (int)(cos(la) * 40), c);
  thickLine(sx, sy, sx + (int)(sin(ra) * -40), sy + (int)(cos(ra) * 40), c);
  int hx = cx - p.lean / 2;
  float ll = radians(p.ll), rl = radians(p.rl);
  thickLine(hx, hipY, hx + (int)(sin(ll) * -50), hipY + (int)(cos(ll) * 50), c);
  thickLine(hx, hipY, hx + (int)(sin(rl) * -50), hipY + (int)(cos(rl) * 50), c);
}

void saverDanceFrame() {
  if (poseDrawn >= 0) drawStick(poseDrawn, C_BG);
  poseIdx = (poseIdx + 1) % POSE_N;
  drawStick(poseIdx, C_SEL);
  poseDrawn = poseIdx;
}

void saverBegin(SaverKind k, uint32_t now) {
  saverKind = k;
  saverSwitchAt = now + SAVER_SWITCH_MS;
  tft.fillScreen(C_BG);
  eyeDrawn = EyeState(); eyeDrawn.open = -1;   // paksa gambar ulang
  poseDrawn = -1;
  nextSaccade = now + 800; blinkAt = now + 2000; blinkPhase = -1;
  txt(W / 2 - 66, H - 14, 1, C_LINE, C_BG, "tekan tombol apa saja");
}

void enterSaver(uint32_t now) {
  mode = M_SAVER;
  saverStart = now;
  saverBegin(SV_EYES, now);
  Serial.println(F("[UI] screensaver ON"));
}

void exitSaver(uint32_t now) {
  mode = M_PAGE;
  page = PG_SUMMARY;
  lastPageChange = now;
  needFull = true;
  Serial.println(F("[UI] screensaver OFF"));
}

void saverLoop(uint32_t now) {
  if (now >= saverSwitchAt) saverBegin(saverKind == SV_EYES ? SV_DANCE : SV_EYES, now);
  uint32_t frameMs = saverKind == SV_EYES ? 50 : 180;
  if (now - saverFrameAt < frameMs) return;
  saverFrameAt = now;
  if (saverKind == SV_EYES) saverEyesFrame(now); else saverDanceFrame();
}

// ============================== NAVIGASI ====================================
void setPage(int p) {
  page = (Page)((p + PG_COUNT) % PG_COUNT);
  lastPageChange = millis();
  needFull = true;
}

void markInput(uint32_t now) {
  lastInputAt = now;
  manualUntil = now + MANUAL_HOLD_MS;
}

void moveCursor(int d) {
  int old = cursor;
  cursor = (cursor + d + CELL_COUNT) % CELL_COUNT;
  if (mode == M_SELECT) { drawCellBox(old, true); drawCellBox(cursor, true); }
  else if (mode == M_DETAIL) drawDetailDynamic();
}

void handleButton(BtnId id, BtnEvt ev, uint32_t heldMs) {
  uint32_t now = millis();
  Button& b = btn[id];

  if (ev == EV_DOWN || ev == EV_LONG)
    Serial.printf("[BTN] %-7s %s t=%lu\n", BTN_NAME[id], ev == EV_DOWN ? "press" : "LONG",
                  (unsigned long)now);

  // Bangun dari screensaver: tombol yang membangunkan "ditelan" sampai dilepas
  if (mode == M_SAVER) {
    if (ev == EV_DOWN) { b.swallow = true; markInput(now); exitSaver(now); }
    return;
  }
  if (b.swallow) return;
  markInput(now);

  switch (id) {
    case B_UP:
    case B_DOWN: {
      int d = (id == B_UP) ? -1 : 1;
      if (ev == EV_DOWN) {
        if (mode == M_PAGE) setPage(page + d);
        else moveCursor(d);
      } else if (ev == EV_REPEAT && mode != M_PAGE) {
        moveCursor(d);
      }
      break;
    }
    case B_ENTER:
      if (ev != EV_DOWN) break;
      if (mode == M_PAGE) {
        if (page == PG_SUMMARY) cursor = bms.minIdx;       // langsung ke cell terendah
        mode = M_SELECT; page = PG_CELLS; needFull = true;
      } else if (mode == M_SELECT) {
        mode = M_DETAIL; needFull = true;
      } else if (mode == M_DETAIL) {
        seenMin[cursor] = seenMax[cursor] = bms.cellMv[cursor];
        drawDetailDynamic();
        toast("Min/max cell ini direset");
      }
      break;
    case B_BACK:
      if (ev != EV_DOWN) break;
      if (mode == M_DETAIL) { mode = M_SELECT; needFull = true; }
      else if (mode == M_SELECT) { mode = M_PAGE; needFull = true; }
      else if (page != PG_SUMMARY) setPage(PG_SUMMARY);
      else toast("Sudah di Summary");
      break;
    case B_RESET:
      if (ev == EV_DOWN) toast("Lepas = reset stat, tahan 3s = + boot count", C_WARN, 3000);
      else if (ev == EV_LONG) {
        bootCount = 0; prefs.putUInt("boots", 0);
        for (int i = 0; i < B_COUNT; i++) btn[i].count = 0;
        resetStats();
        mode = M_PAGE; setPage(PG_SUMMARY);
        toast("Stat + boot count + tombol direset", C_BAD, 2500);
        Serial.println(F("[SYS] full reset counters"));
      } else if (ev == EV_UP && !b.longDone) {
        resetStats();
        mode = M_PAGE; setPage(PG_SUMMARY);
        toast("Statistik min/max direset", C_WARN);
        Serial.println(F("[SYS] stats reset"));
      }
      break;
    case B_RESTART:
      if (ev == EV_DOWN) toast("Tahan 2 detik untuk RESTART...", C_BAD, 2500);
      else if (ev == EV_LONG) {
        tft.fillScreen(C_BG);
        txt(W / 2 - 84, H / 2 - 8, 2, C_BAD, C_BG, "RESTARTING...");
        Serial.println(F("[SYS] restart by button"));
        Serial.flush();
        delay(400);
        ESP.restart();
      } else if (ev == EV_UP && !b.longDone) {
        toast("Restart batal");
      }
      break;
    default: break;
  }
}

// ============================== RENDER LOOP =================================
void renderFull(uint32_t now) {
  tft.fillScreen(C_BG);
  drawHeader();
  drawHeaderDynamic(now);
  drawFooter();
  if (mode == M_DETAIL) { drawDetailStatic(); drawDetailDynamic(); return; }
  switch (page) {
    case PG_SUMMARY: drawSummaryStatic(); drawSummaryDynamic(); break;
    case PG_CELLS:   drawCellsStatic(); break;
    case PG_CHART:   drawChartStatic(); drawChartDynamic(); break;
    case PG_PROT:    drawProtStatic(); break;
    case PG_SYS:     drawSysStatic(); drawSysDynamic(now); drawSysButtons(now, true); break;
    default: break;
  }
}

void renderDynamic(uint32_t now) {
  drawHeaderDynamic(now);
  if (mode == M_DETAIL) { drawDetailDynamic(); return; }
  switch (page) {
    case PG_SUMMARY: drawSummaryDynamic(); break;
    case PG_CELLS:   drawCellsDynamic(); break;
    case PG_CHART:   drawChartDynamic(); break;
    case PG_SYS:     drawSysDynamic(now); break;
    default: break;
  }
}

const char* resetReasonName(esp_reset_reason_t r) {
  switch (r) {
    case ESP_RST_POWERON:   return "POWERON";
    case ESP_RST_EXT:       return "EXT_PIN";
    case ESP_RST_SW:        return "SW_RESTART";
    case ESP_RST_PANIC:     return "PANIC";
    case ESP_RST_INT_WDT:   return "INT_WDT";
    case ESP_RST_TASK_WDT:  return "TASK_WDT";
    case ESP_RST_WDT:       return "WDT";
    case ESP_RST_DEEPSLEEP: return "DEEPSLEEP";
    case ESP_RST_BROWNOUT:  return "BROWNOUT";
    case ESP_RST_SDIO:      return "SDIO";
    default:                return "UNKNOWN";
  }
}

void serialSnapshot(uint32_t now) {
  char up[12]; fmtUptime(up, sizeof(up), now);
  Serial.printf("[%s] V=%.2f I=%.2f SOC=%u%% max=%.3f(#%d) min=%.3f(#%d) d=%umV heap=%u\n",
                up, bms.packV, bms.current, bms.soc,
                bms.maxMv / 1000.0, bms.maxIdx + 1, bms.minMv / 1000.0, bms.minIdx + 1,
                bms.maxMv - bms.minMv, ESP.getFreeHeap());
}

void setup() {
  Serial.begin(115200);
  delay(50);

  for (int i = 0; i < B_COUNT; i++) pinMode(BTN_PIN[i], INPUT_PULLUP);
  if (TFT_BL >= 0) { pinMode(TFT_BL, OUTPUT); digitalWrite(TFT_BL, HIGH); }

  resetReasonStr = resetReasonName(esp_reset_reason());
  prefs.begin("bmsdummy", false);
  bootCount = prefs.getUInt("boots", 0) + 1;
  prefs.putUInt("boots", bootCount);

  randomSeed(esp_random());
  resetStats();
  tickData();

  tft.begin(TFT_SPI_HZ);
  tft.setRotation(TFT_ROTATION);
  tft.cp437(true);
  W = tft.width(); H = tft.height();

  Serial.printf("\n[BOOT] BMS dummy display | boot #%lu | reset=%s | %dx%d\n",
                (unsigned long)bootCount, resetReasonStr, W, H);

  uint32_t now = millis();
  lastInputAt = now; lastPageChange = now; manualUntil = 0;
  // baca keadaan awal tombol supaya tombol yang tertahan saat boot tidak dianggap tekan
  for (int i = 0; i < B_COUNT; i++) {
    btn[i].raw = btn[i].stable = digitalRead(BTN_PIN[i]) == LOW;
    btn[i].swallow = btn[i].stable;
    btn[i].longDone = true;
  }
  needFull = true;
}

void loop() {
  static uint32_t lastData = 0, lastUi = 0, lastLog = 0;
  uint32_t now = millis();

  pollButtons(now);

  if (now - lastData >= DATA_TICK_MS) { lastData = now; tickData(); }
  if (now - lastLog >= SERIAL_LOG_MS) { lastLog = now; serialSnapshot(now); }

  if (mode == M_SAVER) { saverLoop(now); delay(2); return; }

  if (SCREENSAVER_MS > 0 && now - lastInputAt >= SCREENSAVER_MS) {
    enterSaver(now);
    return;
  }

  if (mode == M_PAGE && PAGE_INTERVAL_MS > 0 && now >= manualUntil &&
      now - lastPageChange >= PAGE_INTERVAL_MS) {
    setPage(page + 1);
  }

  if (needFull) {
    needFull = false;
    renderFull(now);
    lastUi = now;
    if (toastUntil > now) toast(toastMsg, toastColor, toastUntil - now);
  } else if (now - lastUi >= UI_REFRESH_MS) {
    lastUi = now;
    renderDynamic(now);
  } else if (page == PG_SYS && mode == M_PAGE) {
    static uint32_t lastSys = 0;           // tes tombol lebih responsif
    if (now - lastSys >= 30) { lastSys = now; drawSysButtons(now, false); }
  }

  if (toastShown && now >= toastUntil) drawFooter();
  delay(2);
}
